async function generateAI(){
 const target=current;if(!target.name?.trim()||(target.purpose||'').trim().length<15){show('<h3>Önce fikrinizi yazın</h3><p>Proje adı ile amaç alanına en az 15 karakterlik bir fikir açıklaması girin.</p><div class="actions"><button class="primary" onclick="closeModal()">Tamam</button></div>');return}
 const button=el('generateAI');button.disabled=true;button.textContent='Gemini projenizi hazırlıyor…';
 try{
  const response=await fetch('/api/generate-project',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({project:target}),signal:AbortSignal.timeout(100000)});
  let data;try{data=await response.json()}catch{throw Error('Gemini sunucusu çalışmıyor. Gemini ile başlat dosyasını kullanın.')}
  if(!response.ok)throw Error(data.error||'Proje hazırlanamadı.');
  const names={purpose:'Amaç',teamIntro:'Takım tanıtımı',summary:'Özet',problem:'Problem / giriş',solution:'Çözüm',method:'Yöntem',prototype:'Prototip',sustainability:'Sürdürülebilirlik',originality:'Özgünlük',audience:'Hedef kitle',evaluation:'Başarı ölçütleri',strengths:'Güçlü yönler',weaknesses:'Zayıf yönler',opportunities:'Fırsatlar',threats:'Tehditler',suggestions:'Öneriler',keywords:'Anahtar kelimeler'};
  show(`<h3>Gemini proje önerisi</h3><p>Öneriyi inceleyin. Uyguladığınızda mevcut metnin bir yedeği bu tarayıcıda tutulur. Kaynaklar ve gerçekleşen bulgular değiştirilmez.</p><div style="max-height:55vh;overflow:auto">${Object.entries(data.sections).filter(([k,v])=>names[k]&&v).map(([k,v])=>`<h4>${names[k]}</h4><p style="white-space:pre-wrap">${safe(v)}</p>`).join('')}<h4>Kontrol edilecek noktalar</h4><ul>${data.reviewNotes.map(n=>`<li>${safe(n)}</li>`).join('')}</ul></div><div class="actions"><button class="secondary" onclick="closeModal()">Vazgeç</button><button id="applyAI" class="primary">Öneriyi projeme uygula</button></div>`);
  el('applyAI').onclick=()=>{const previous={};for(const k of Object.keys(names))previous[k]=target[k]||'';target.aiHistory=[{date:new Date().toISOString(),sections:previous},...(target.aiHistory||[])].slice(0,3);for(const [k,v]of Object.entries(data.sections))if(names[k]&&typeof v==='string'&&v.trim())target[k]=v;target.aiModel=data.model;target.aiReviewNotes=data.reviewNotes;target.aiDeclaration=`Kullanılan araç: Google Gemini ${data.model}. Erişim tarihi: ${new Date().toLocaleDateString('tr-TR')}. Kullanım amacı: proje önerisi ve metin taslağı hazırlanması. Kapsam: proje açıklama bölümleri. İnsan müdahalesi ve doğrulama: öğretmen tarafından kontrol edilip tamamlanacaktır.`;persist();closeModal();if(current===target){step=5;display()}};
 }catch(e){show(`<h3>Gemini bağlantısı</h3><p>${safe(e.name==='TimeoutError'?'İstek süresi doldu. Daha sonra yeniden deneyin.':e.message)}</p><p>Yeni anahtarınızı sohbete yazmadan <b>Gemini-ile-baslat.ps1</b> dosyasının gizli girişine ekleyin.</p><div class="actions"><button class="primary" onclick="closeModal()">Kapat</button></div>`)}finally{if(el('generateAI')){el('generateAI').disabled=false;el('generateAI').textContent='✧ Gemini ile proje hazırla'}}
}
function restoreAI(){const previous=current.aiHistory?.shift();if(!previous)return;Object.assign(current,previous.sections);persist();display()}
